#include <stdio.h>

int main() {
    int num;
    int sum = 0;

    // Read the five-digit integer
    scanf("%d", &num);

    // Loop to extract and sum each digit
    while (num > 0) {
        sum += num % 10; // Get the last digit
        num /= 10;       // Remove the last digit
    }

    // Print the final sum
    printf("%d\n", sum);

    return 0;
}
