#include <stdio.h>

int main() {
    char ch;
    char s[100];
    char sen[100];

    // Read the character and consume its trailing newline
    scanf("%c", &ch);
    
    // Read the single word string and consume its trailing newline
    scanf("%s", s);
    
    // Clean up the leftover newline character in the input buffer
    scanf("\n");
    
    // Read the complete sentence including spaces until a newline is hit
    scanf("%[^\n]%*c", sen);

    // Print the outputs in sequence
    printf("%c\n", ch);
    printf("%s\n", s);
    printf("%s\n", sen);

    return 0;
}
