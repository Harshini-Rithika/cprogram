#include <stdio.h>
#include <stdlib.h>
typedef struct {
    int index;
    long long diff_val; 
} DequeNode;
long long findMaxValueOfEquation(int** points, int pointsSize, int* pointsColSize, int k) {
    DequeNode* deque = (DequeNode*)malloc(pointsSize * sizeof(DequeNode));
    int head = 0;
    int tail = 0;
    long long max_result = -4e18; 
    for (int j = 0; j < pointsSize; j++) {
        long long xj = points[j][0];
        long long yj = points[j][1];
        while (head < tail && (xj - points[deque[head].index][0]) > k) {
            head++;
        }
        if (head < tail) {
            long long current_val = deque[head].diff_val + yj + xj;
            if (current_val > max_result) {
                max_result = current_val;
            }
        }
        long long current_diff = yj - xj;
        while (head < tail && deque[tail - 1].diff_val <= current_diff) {
            tail--;
        }
        deque[tail].index = j;
        deque[tail].diff_val = current_diff;
        tail++;
    }
    free(deque);
    return max_result;
}
