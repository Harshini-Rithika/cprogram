#include <stdio.h>
#include <string.h>
#include <ctype.h>
int main() {
    char s[10005];
    if (scanf("%s", s) != 1) {
        return 0;
    }
    int frequency[10] = {0};
    for (int i = 0; s[i] != '\0'; i++) {
        if (s[i] >= '0' && s[i] <= '9') {
            frequency[s[i] - '0']++;
        }
    }
    for (int i = 0; i < 10; i++) {
        printf("%d", frequency[i]);
        if (i < 9) {
            printf(" "); 
        }
    }
    printf("\n");
    
    return 0;
}
