#include <string.h>
#include <stdlib.h>
char* simplifyPath(char* path) {
    int len = strlen(path);
    char** stack = (char**)malloc(len * sizeof(char*));
    int top = 0;
    char* portion = strtok(path, "/");
    while (portion != NULL) {
        if (strcmp(portion, "..") == 0) {
            if (top > 0) {
                top--;
            }
        } else if (strcmp(portion, ".") == 0 || strcmp(portion, "") == 0) {
        } else {
            stack[top++] = portion;
        }
        portion = strtok(NULL, "/");
    }
    char* result = (char*)malloc((len + 2) * sizeof(char));
    result[0] = '\0'; 
    if (top == 0) {
        strcpy(result, "/");
    } else {
        for (int i = 0; i < top; i++) {
            strcat(result, "/");
            strcat(result, stack[i]);
        }
    }
    free(stack);
    return result;
}
