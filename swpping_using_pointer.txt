 #include<stdio.h>
  void swap(int *,int *);
  int main()
  {
     int a,b;
     printf("enter any two numbers:");
     scanf("%d %d",&a,&b);
     printf("before swapping a=%d and b=%d",a,b);
     swap(&a,&b);
     printf("after swapping a=%d and b=%d",a,b);
     return 0;
  }
  void swap(int *p,int *q)
  {
     int temp;
     temp=*p;
     *p=*q;
     *q=temp;
  }
