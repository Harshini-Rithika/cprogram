int* maxSlidingWindow(int* nums, int numsSize, int k, int* returnSize) {
    if (numsSize == 0 || k == 0) {
        *returnSize = 0;
        return NULL;
    }
    int* res = (int*)malloc((numsSize - k + 1) * sizeof(int));
    *returnSize = numsSize - k + 1;
    int deque[numsSize];
    int head = 0, tail = -1;
    int idx = 0;
    for (int i = 0; i < numsSize; i++) {
        if (head <= tail && deque[head] < i - k + 1) {
            head++;
        }
        while (head <= tail && nums[deque[tail]] < nums[i]) {
            tail--;
        }
        deque[++tail] = i;
        if (i >= k - 1) {
            res[idx++] = nums[deque[head]];
        }
    }  
    return res;
}
